function quydao_mp_singularity_multi
% 3-RRR SYMMETRIC PLANAR PARALLEL MANIPULATOR
clc; clear; close all;

%% ========== 1. THAM SO ==========
H          = 50;                             % canh tam giac co so (cm)
l          = 20;                             % chieu dai moi khau  (cm)
m_plat     = (sqrt(3)/30)*H;                 % G den dinh platform (cm)
KL         = 100;                            % khoi luong platform (g)
m_seg      = 100;                            % khoi luong moi khau (g)
w_arm      = 4;                              % rong khau (cm)
t_arm      = 0.5;                            % day khau (cm)
g_acc      = 981;                            % gia toc trong truong (cm/s^2)

I_link     = (1/3)*(m_seg/1000)*(l/100)^2;   % kg.m^2
tau_max    = 0.6;                            % gioi han moment dong co (N.m)
rpm_max_adj = 112 * sqrt(100/KL);            % gioi han RPM hieu chinh

fprintf('=== THONG SO ROBOT 3-RRR ===\n');
fprintf('Khau: 2 x %.0f cm | R=%.1f cm | Day=%.1f cm\n', l, w_arm, t_arm);
fprintf('Khoi luong: Platform=%g g | Moi khau=%g g\n', KL, m_seg);
fprintf('I_link=%.6f kg.m^2 | RPM_max=%.1f\n', I_link, rpm_max_adj);

%% ========== 2. TOA DO CO SO ==========
A1 = [0, 0];
A2 = [H, 0];
A3 = [H/2, (sqrt(3)/2)*H];
A  = [A1; A2; A3];

%% ========== 3. CHON QUY DAO ==========
fprintf('\n1=Duong thang  2=Tam giac  3=Hinh vuong  4=Duong tron\n');
traj_choice = input('Chon quy dao (1-4): ');
x0 = H/2;
y0 = (sqrt(3)/4)*H;
n  = 100;

switch traj_choice
    case 1
        v_des  = input('Van toc (cm/s): ');
        xt = input('X dich (cm): ');
        yt = input('Y dich (cm): ');
        dist   = sqrt((xt-x0)^2 + (yt-y0)^2);
        T_sim  = dist / v_des;
        x_traj = linspace(x0, xt, n);
        y_traj = linspace(y0, yt, n);
    case 2
        v_des = input('Van toc (cm/s): ');
        sl    = input('Canh tam giac (cm): ');
        ang   = [90,210,330]*pi/180;
        r_tri = sl / sqrt(3);
        nps   = floor(n/3);
        x_traj = []; y_traj = [];
        for i = 1:3
            ni = mod(i,3)+1;
            x_traj = [x_traj; linspace(x0+r_tri*cos(ang(i)), x0+r_tri*cos(ang(ni)), nps)'];
            y_traj = [y_traj; linspace(y0+r_tri*sin(ang(i)), y0+r_tri*sin(ang(ni)), nps)'];
        end
        n = length(x_traj); dist = 3*sl; T_sim = dist/v_des;
    case 3
        v_des = input('Van toc (cm/s): ');
        sl    = input('Canh hinh vuong (cm): ');
        hs    = sl/2;
        corn  = [x0-hs,y0-hs; x0+hs,y0-hs; x0+hs,y0+hs; x0-hs,y0+hs];
        nps   = floor(n/4);
        x_traj = []; y_traj = [];
        for i = 1:4
            ni = mod(i,4)+1;
            x_traj = [x_traj; linspace(corn(i,1), corn(ni,1), nps)'];
            y_traj = [y_traj; linspace(corn(i,2), corn(ni,2), nps)'];
        end
        n = length(x_traj); dist = 4*sl; T_sim = dist/v_des;
    case 4
        v_des = input('Van toc (cm/s): ');
        rc    = input('Ban kinh (cm): ');
        thc   = linspace(0, 2*pi, n);
        x_traj = x0 + rc*cos(thc);
        y_traj = y0 + rc*sin(thc);
        dist = 2*pi*rc; T_sim = dist/v_des;
    otherwise
        error('Lua chon khong hop le');
end
phi_traj   = zeros(n, 1);
traj_names = {'Duong thang','Tam giac','Hinh vuong','Duong tron'};
fprintf('Quy dao: %s | L=%.2f cm | T=%.2f s | n=%d\n', ...
    traj_names{traj_choice}, dist, T_sim, n);

%% ========== 4. KIEM TRA KHONG GIAN LAM VIEC ==========
r_max = 2*l + m_plat;
for k = 1:n
    d1 = norm([x_traj(k), y_traj(k)]);
    d2 = norm([x_traj(k)-H/2, y_traj(k)-(sqrt(3)/2)*H]);
    d3 = norm([x_traj(k)-H,   y_traj(k)]);
    if d1>r_max || d2>r_max || d3>r_max
        error('Buoc %d: Ngoai khong gian lam viec!', k);
    end
end
fprintf('Kiem tra khong gian: OK\n');

%% ========== 5. BAN DO KY DI ==========
fprintf('Dang tinh ban do ky di...\n');
[X_grid, Y_grid] = meshgrid(linspace(-5,H+5,150), linspace(-5,H+5,150));
det_map  = zeros(size(X_grid));
sing_thr = 1e-3;
cond_thr = 100;

for ii = 1:size(X_grid,1)
    for jj = 1:size(X_grid,2)
        xP = X_grid(ii,jj); yP = Y_grid(ii,jj);
        Pg = [xP+m_plat*cos(210*pi/180), yP+m_plat*sin(210*pi/180);
              xP+m_plat*cos(330*pi/180), yP+m_plat*sin(330*pi/180);
              xP+m_plat*cos( 90*pi/180), yP+m_plat*sin( 90*pi/180)];
        Jg  = zeros(3,3); vld = true;
        for k = 1:3
            dx = Pg(k,1)-A(k,1); dy = Pg(k,2)-A(k,2);
            d  = sqrt(dx^2+dy^2);
            if d>2*l || d<0.01, vld=false; break; end
            th = atan2(dy,dx) - acos(d/(2*l));
            Jg(k,1) = -l*sin(th); Jg(k,2) = l*cos(th);
            Jg(k,3) = -l*sin(th)*(Pg(k,2)-yP) + l*cos(th)*(Pg(k,1)-xP);
        end
        if vld
            dJ = abs(det(Jg)); det_map(ii,jj) = dJ;
        else
            det_map(ii,jj) = NaN;
        end
    end
end
fprintf('Ban do ky di: Hoan thanh\n');

%% ========== 6. HINH VE TONG HOP -- 2D + 3D + sin/cos + Moment ==========
fig1 = figure('Name','Mo Phong Robot 3-RRR', 'NumberTitle','off', ...
    'Units','normalized', 'OuterPosition',[0 0 1 1], 'Color','white');

% ---- Bo cuc  [trai  duoi  rong  cao] ----
ax_2d  = axes('Position',[0.02  0.50  0.16  0.44]);   % Ban do 2D ky di
ax_inf = axes('Position',[0.02  0.02  0.16  0.45]);   % Bang thong so
ax_3d  = axes('Position',[0.21  0.34  0.29  0.57]);   % Mo phong 3D
ax_sin = axes('Position',[0.21  0.05  0.13  0.25]);   % sin(theta)
ax_cos = axes('Position',[0.36  0.05  0.13  0.25]);   % cos(theta)

% Moment 2x2
ax_tor = gobjects(4,1);
tor_pos = [0.52  0.54  0.23  0.37;
           0.77  0.54  0.22  0.37;
           0.52  0.08  0.23  0.37;
           0.77  0.08  0.22  0.37];
for sp = 1:4
    ax_tor(sp) = axes('Position', tor_pos(sp,:));
end

% --- Ban do 2D ky di ---
axes(ax_2d); hold on; axis equal; grid on;
axis([-10 H+10 -10 H+10]);
xlabel('X (cm)'); ylabel('Y (cm)');
title('Dinh thuc Jacobian (log10)', 'FontSize',11, 'FontWeight','bold');
contourf(ax_2d, X_grid, Y_grid, log10(det_map+1e-10), 20, 'LineColor','none');
cb = colorbar; colormap(ax_2d, jet); clim([-3 1]);
cb.Label.String = 'log10|det J|';
contour(ax_2d, X_grid, Y_grid, det_map, [sing_thr sing_thr], 'r-', 'LineWidth',2);
plot(ax_2d, [A1(1) A2(1) A3(1) A1(1)], [A1(2) A2(2) A3(2) A1(2)], 'k-','LineWidth',3);
text(ax_2d, A1(1)-3, A1(2)-3, 'A1','FontSize',10,'FontWeight','bold');
text(ax_2d, A2(1)+2, A2(2)-3, 'A2','FontSize',10,'FontWeight','bold');
text(ax_2d, A3(1),   A3(2)+2, 'A3','FontSize',10,'FontWeight','bold');
plot(ax_2d, x_traj, y_traj, 'w-','LineWidth',2);
plot(ax_2d, x_traj(1),   y_traj(1),   'go','MarkerSize',10,'MarkerFaceColor','g','LineWidth',2);
plot(ax_2d, x_traj(end), y_traj(end), 'r*','MarkerSize',15,'LineWidth',2);
pt_G2d    = plot(ax_2d, nan, nan, 'wo','MarkerFaceColor','w','MarkerSize',8,'LineWidth',2);
pt_plat2d = fill(ax_2d, nan, nan, 'c','FaceAlpha',0.5,'EdgeColor','w','LineWidth',2);
h_link2d  = gobjects(3,1);
for i = 1:3
    h_link2d(i) = plot(ax_2d, nan, nan, 'y-','LineWidth',3);
end

% --- Mo phong 3D ---
axes(ax_3d); hold on; axis equal; grid on;
xlabel('X (cm)','FontSize',12);
ylabel('Y (cm)','FontSize',12);
zlabel('Z (cm)','FontSize',12);
title('Mo Phong 3D Robot', 'FontSize',14,'FontWeight','bold');
view(45,35); lighting gouraud; camlight('right'); camlight('left');
set(ax_3d,'Color','white', ...
    'XColor',[0.2 0.2 0.2],'YColor',[0.2 0.2 0.2],'ZColor',[0.2 0.2 0.2], ...
    'GridColor',[0.75 0.75 0.75],'GridAlpha',0.5,'Box','on');
axis(ax_3d, [-10 H+10 -10 H+10 -8 8]);

% Nen san
[bx,by] = meshgrid(linspace(-5,H+5,2), linspace(-5,H+5,2));
surf(ax_3d, bx, by, zeros(2)-2, ...
    'FaceColor',[0.90 0.90 0.92],'EdgeColor','none','FaceAlpha',0.6);
fill3(ax_3d, [A1(1) A2(1) A3(1) A1(1)], [A1(2) A2(2) A3(2) A1(2)], zeros(1,4)-1.5, ...
    [0.75 0.75 0.78],'FaceAlpha',0.85,'EdgeColor',[0.2 0.2 0.2],'LineWidth',2);

hw = w_arm/2;  ht = t_arm/2;  n_cap = 14;  jr = hw*0.7;
arm_col = [0.92 0.28 0.10;
           0.10 0.72 0.28;
           0.15 0.45 0.92];

h3d_links  = gobjects(3,2);
h3d_caps1  = gobjects(3,2);
h3d_caps2  = gobjects(3,2);
h3d_joints = gobjects(3,2);

h3d_plat = patch(ax_3d,'XData',[],'YData',[],'ZData',[], ...
    'FaceColor',[0.0 0.65 0.85],'FaceAlpha',0.88, ...
    'EdgeColor',[0.0 0.3 0.6],'LineWidth',2);
h3d_ctr  = plot3(ax_3d, nan,nan,nan, 'o', ...
    'MarkerFaceColor',[0.0 0.3 0.6],'MarkerEdgeColor','w','MarkerSize',10,'LineWidth',2);

[sx,sy,sz] = sphere(16);
for i = 1:3
    col  = arm_col(i,:);
    cold = col * 0.72;
    [cx_t,cy_t,cz_t] = arm_body_surf([0 0 0],[1 0 0],hw,ht);
    h3d_links(i,1)  = surf(ax_3d,cx_t,cy_t,cz_t,'FaceColor',col, 'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.5,'AmbientStrength',0.4);
    h3d_links(i,2)  = surf(ax_3d,cx_t,cy_t,cz_t,'FaceColor',cold,'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.5,'AmbientStrength',0.4);
    [cx_t,cy_t,cz_t] = arm_cap_arc([0 0 0],[1 0 0],-1,hw,ht,n_cap);
    h3d_caps1(i,1) = surf(ax_3d,cx_t,cy_t,cz_t,'FaceColor',col, 'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.5,'AmbientStrength',0.4);
    h3d_caps1(i,2) = surf(ax_3d,cx_t,cy_t,cz_t,'FaceColor',cold,'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.5,'AmbientStrength',0.4);
    [cx_t,cy_t,cz_t] = arm_cap_arc([0 0 0],[1 0 0],+1,hw,ht,n_cap);
    h3d_caps2(i,1) = surf(ax_3d,cx_t,cy_t,cz_t,'FaceColor',col, 'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.5,'AmbientStrength',0.4);
    h3d_caps2(i,2) = surf(ax_3d,cx_t,cy_t,cz_t,'FaceColor',cold,'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.5,'AmbientStrength',0.4);
    h3d_joints(i,1) = surf(ax_3d,sx*jr,sy*jr,sz*jr,'FaceColor',[0.88 0.88 0.90],'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.8);
    h3d_joints(i,2) = surf(ax_3d,sx*jr,sy*jr,sz*jr,'FaceColor',[0.78 0.78 0.80],'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.8);
    surf(ax_3d, sx*jr*1.3+A(i,1), sy*jr*1.3+A(i,2), sz*jr*1.3, ...
        'FaceColor',col,'EdgeColor','none','FaceAlpha',1,'SpecularStrength',0.6);
end
axes(ax_inf); axis off;

%% ========== 7. TINH TRUOC THETA, SIN, COS ==========
theta_pre = zeros(n, 3);
for k = 1:n
    xP = x_traj(k); yP = y_traj(k); phi = phi_traj(k);
    Pp = [xP+m_plat*cos(phi+210*pi/180), yP+m_plat*sin(phi+210*pi/180);
          xP+m_plat*cos(phi+330*pi/180), yP+m_plat*sin(phi+330*pi/180);
          xP+m_plat*cos(phi+ 90*pi/180), yP+m_plat*sin(phi+ 90*pi/180)];
    for i = 1:3
        dx = Pp(i,1)-A(i,1); dy = Pp(i,2)-A(i,2);
        d  = sqrt(dx^2+dy^2);
        if d <= 2*l && d > 0.01
            theta_pre(k,i) = atan2(dy,dx) - acos(d/(2*l));
        end
    end
end
sin_pre = sin(theta_pre);
cos_pre = cos(theta_pre);
T_vec   = linspace(0, T_sim, n);

%% ========== 8. BIEU DO SIN/COS + MOMENT ==========
anim_col = [0.85 0.10 0.10;
            0.10 0.65 0.20;
            0.10 0.35 0.85];

% --- Bieu do sin(theta) ---
hold(ax_sin,'on'); grid(ax_sin,'on'); box(ax_sin,'on');
title(ax_sin, 'sin(\theta) cac khop', 'FontSize',10,'FontWeight','bold');
xlabel(ax_sin, 'Thoi gian (s)', 'FontSize',9);
ylabel(ax_sin, 'sin(\theta)', 'FontSize',9);
axis(ax_sin, [0 T_sim -1.15 1.15]);
yline(ax_sin, 1.0,'--','Color',[0.80 0.80 0.80],'LineWidth',0.8,'HandleVisibility','off');
yline(ax_sin,-1.0,'--','Color',[0.80 0.80 0.80],'LineWidth',0.8,'HandleVisibility','off');
yline(ax_sin, 0.0, '-','Color',[0.88 0.88 0.88],'LineWidth',0.6,'HandleVisibility','off');
for i = 1:3
    plot(ax_sin, T_vec, sin_pre(:,i), '-', 'Color', anim_col(i,:)*0.35+0.65, ...
        'LineWidth',0.9, 'DisplayName', sprintf('Khop %d',i));
end
legend(ax_sin,'Location','best','FontSize',7,'Box','off');
h_sin_pt = gobjects(3,1);
for i = 1:3
    h_sin_pt(i) = plot(ax_sin, T_vec(1), sin_pre(1,i), 'o', ...
        'MarkerFaceColor',anim_col(i,:),'MarkerEdgeColor','w', ...
        'MarkerSize',7,'LineWidth',1.5,'HandleVisibility','off');
end

% --- Bieu do cos(theta) ---
hold(ax_cos,'on'); grid(ax_cos,'on'); box(ax_cos,'on');
title(ax_cos, 'cos(\theta) cac khop', 'FontSize',10,'FontWeight','bold');
xlabel(ax_cos, 'Thoi gian (s)', 'FontSize',9);
ylabel(ax_cos, 'cos(\theta)', 'FontSize',9);
axis(ax_cos, [0 T_sim -1.15 1.15]);
yline(ax_cos, 1.0,'--','Color',[0.80 0.80 0.80],'LineWidth',0.8,'HandleVisibility','off');
yline(ax_cos,-1.0,'--','Color',[0.80 0.80 0.80],'LineWidth',0.8,'HandleVisibility','off');
yline(ax_cos, 0.0, '-','Color',[0.88 0.88 0.88],'LineWidth',0.6,'HandleVisibility','off');
for i = 1:3
    plot(ax_cos, T_vec, cos_pre(:,i), '--', 'Color', anim_col(i,:)*0.35+0.65, ...
        'LineWidth',0.9, 'DisplayName', sprintf('Khop %d',i));
end
legend(ax_cos,'Location','best','FontSize',7,'Box','off');
h_cos_pt = gobjects(3,1);
for i = 1:3
    h_cos_pt(i) = plot(ax_cos, T_vec(1), cos_pre(1,i), 's', ...
        'MarkerFaceColor',anim_col(i,:),'MarkerEdgeColor','w', ...
        'MarkerSize',7,'LineWidth',1.5,'HandleVisibility','off');
end

%% ========== 8b. BIEU DO MOMENT ==========
tau_mNm = tau_max * 1000;
col3    = anim_col;
tvec    = T_vec;

sp_title = {'Moment quan tinh  I*alpha', 'Moment trong - Khau', ...
            'Moment trong - Platform',   'Tong moment & Gioi han'};
sp_ylab  = {'tau_{qt} (mN.m)', 'tau_{tkh} (mN.m)', ...
            'tau_{tpl} (mN.m)', 'tau_{tong} (mN.m)'};

pt_tor = gobjects(4,3);
ln_tor = gobjects(4,3);

for sp = 1:4
    hold(ax_tor(sp),'on'); grid(ax_tor(sp),'on'); box(ax_tor(sp),'on');
    title(ax_tor(sp), sp_title{sp}, 'FontSize',9,'FontWeight','bold');
    xlabel(ax_tor(sp), 'Thoi gian (s)', 'FontSize',8);
    ylabel(ax_tor(sp), sp_ylab{sp}, 'FontSize',8);
    set(ax_tor(sp),'FontSize',8);
    for i = 1:3
        ln_tor(sp,i) = plot(ax_tor(sp), nan, nan, '-', ...
            'Color',col3(i,:),'LineWidth',1.5,'DisplayName',sprintf('Khop %d',i));
    end
    if sp == 4
        yline(ax_tor(sp), tau_mNm, 'r--', 'LineWidth',1.5, ...
            'DisplayName', sprintf('max=%.0f mN.m',tau_mNm));
        text(ax_tor(sp), tvec(max(1,round(n*0.5))), tau_mNm*1.06, 'QUA TAI', ...
            'Color',[0.8 0 0],'FontSize',7,'FontWeight','bold','HorizontalAlignment','center');
    end
    legend(ax_tor(sp),'Location','best','FontSize',7,'Box','off');
    for i = 1:3
        pt_tor(sp,i) = plot(ax_tor(sp), nan, nan, 'o', ...
            'MarkerFaceColor',col3(i,:),'MarkerEdgeColor','w', ...
            'MarkerSize',7,'LineWidth',1.2,'HandleVisibility','off');
    end
end

% Tieu de tong the hinh 1
sgtitle(fig1, sprintf('Mo Phong Robot 3-RRR  |  Quy dao: %s  |  v=%.1f cm/s  |  T=%.2f s', ...
    traj_names{traj_choice}, v_des, T_sim), 'FontSize',13,'FontWeight','bold');

%% ========== 9. MANG DU LIEU ==========
theta_all        = zeros(n,3);
det_J_all        = zeros(n,1);
cond_J_all       = zeros(n,1);
singularity_flag = zeros(n,1);
near_sing_flag   = zeros(n,1);
v_mag            = zeros(n,1);
rpm_all          = zeros(n,3);
torque_all       = zeros(n,3);
torque_inertia   = zeros(n,3);
torque_grav_link = zeros(n,3);
torque_grav_plat = zeros(n,3);
rpm_over_flag    = zeros(n,3);
dt               = T_sim / (n-1);

%% ========== 10. VONG LAP MO PHONG ==========
fprintf('\n=== BAT DAU MO PHONG ===\n');
for k = 1:n
    xP  = x_traj(k); yP  = y_traj(k); phi = phi_traj(k);
    t_k = (k-1) * dt;

    P1 = [xP+m_plat*cos(phi+210*pi/180), yP+m_plat*sin(phi+210*pi/180)];
    P2 = [xP+m_plat*cos(phi+330*pi/180), yP+m_plat*sin(phi+330*pi/180)];
    P3 = [xP+m_plat*cos(phi+ 90*pi/180), yP+m_plat*sin(phi+ 90*pi/180)];
    P  = [P1; P2; P3];

    theta  = zeros(1,3);
    J_kin  = zeros(3,3);
    J_pos  = zeros(3,2);
    ok     = true;

    for i = 1:3
        dx = P(i,1)-A(i,1); dy = P(i,2)-A(i,2);
        d  = sqrt(dx^2+dy^2);
        if d>2*l || d<0.01, ok=false; break; end
        al = atan2(dy,dx); be = acos(d/(2*l));
        theta(i)  = al - be;
        J_pos(i,:) = [A(i,1)+l*cos(theta(i)), A(i,2)+l*sin(theta(i))];
        J_kin(i,1) = -l*sin(theta(i));
        J_kin(i,2) =  l*cos(theta(i));
        J_kin(i,3) = -l*sin(theta(i))*(P(i,2)-yP) + l*cos(theta(i))*(P(i,1)-xP);
    end
    if ~ok, continue; end

    % Ket qua dong hoc
    theta_all(k,:) = theta;
    dJ = abs(det(J_kin)); det_J_all(k) = dJ;
    cond_J_all(k)  = ternval(dJ>1e-10, cond(J_kin), Inf);
    if dJ < sing_thr,            singularity_flag(k) = 1; end
    if cond_J_all(k) > cond_thr, near_sing_flag(k)  = 1; end

    % Van toc
    if k>1 && k<n
        vx = (x_traj(k+1)-x_traj(k-1))/(2*dt);
        vy = (y_traj(k+1)-y_traj(k-1))/(2*dt);
    elseif k==1
        vx = (x_traj(2)-x_traj(1))/dt;
        vy = (y_traj(2)-y_traj(1))/dt;
    else
        vx = (x_traj(end)-x_traj(end-1))/dt;
        vy = (y_traj(end)-y_traj(end-1))/dt;
    end
    v_mag(k) = sqrt(vx^2+vy^2);

    % RPM + Moment
    if k > 1
        omega = (theta_all(k,:)-theta_all(k-1,:)) / dt;
        rpm_all(k,:) = abs(omega)*60/(2*pi);
        if k > 2
            alpha_a = (omega - (theta_all(k-1,:)-theta_all(k-2,:))/dt) / dt;
            for i = 1:3
                torque_inertia(k,i)    = I_link * abs(alpha_a(i));

%% ========== THAN KHAU (dai chu nhat) ==========
function [Xc,Yc,Zc] = arm_body_surf(p1, p2, hw, ht)
    vec = p2-p1;  len = norm(vec);
    if len < 1e-6, Xc=zeros(2,5); Yc=Xc; Zc=Xc; return; end
    ax_l = vec/len;
    ax_w = cross([0 0 1], ax_l);
    if norm(ax_w)<1e-6, ax_w=[1 0 0]; else, ax_w=ax_w/norm(ax_w); end
    ax_t = cross(ax_l, ax_w);  ax_t = ax_t/norm(ax_t);
    cs = [hw,ht; hw,-ht; -hw,-ht; -hw,ht; hw,ht];
    Xc = zeros(2,5);  Yc = Xc;  Zc = Xc;
    for ii = 1:2
        pt = ternval(ii==1, p1, p2);
        for jj = 1:5
            off = cs(jj,1)*ax_w + cs(jj,2)*ax_t;
            Xc(ii,jj)=pt(1)+off(1); Yc(ii,jj)=pt(2)+off(2); Zc(ii,jj)=pt(3)+off(3);
        end
    end
end


%% ========== NUT KHAU (ban nguyet) ==========
function [Xc,Yc,Zc] = arm_cap_arc(p1, p2, side, hw, ht, n_cap)
    vec = p2-p1;  len = norm(vec);
    if len < 1e-6, Xc=zeros(n_cap,5); Yc=Xc; Zc=Xc; return; end
    ax_l = vec/len;
    ax_w = cross([0 0 1], ax_l);
    if norm(ax_w)<1e-6, ax_w=[1 0 0]; else, ax_w=ax_w/norm(ax_w); end
    ax_t = cross(ax_l, ax_w);  ax_t = ax_t/norm(ax_t);
    if side < 0, ctr=p1; dir=-ax_l; else, ctr=p2; dir=ax_l; end
    cs = [hw,ht; hw,-ht; -hw,-ht; -hw,ht; hw,ht];
    phi_v = linspace(-pi/2, pi/2, n_cap);
    Xc = zeros(n_cap,5);  Yc = Xc;  Zc = Xc;
    for ii = 1:n_cap
        phi = phi_v(ii);
        pt  = ctr + dir*(hw*sin(phi));
        for jj = 1:5
            off = cs(jj,1)*cos(phi)*ax_w + cs(jj,2)*ax_t;
            Xc(ii,jj)=pt(1)+off(1); Yc(ii,jj)=pt(2)+off(2); Zc(ii,jj)=pt(3)+off(3);
        end
    end
end


%% ========== HAM TIEN ICH ==========
function r = ternval(cond, tv, fv)
    if cond, r=tv; else, r=fv; end
end
