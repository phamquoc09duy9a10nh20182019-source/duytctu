function PID_DongLucHoc
% =========================================================
% BỘ ĐIỀU KHIỂN PID ĐỘNG LỰC HỌC BẬC 2 - ROBOT 3-RRR
% Computed Torque Control (CTC) vs PID thường
%
% [DO DAC]  = cần đo từ phần cứng thực tế
% [TUNING]  = chỉnh sau khi có kết quả thực tế
% [GIU NGUYEN] = lấy từ file Matlab.txt gốc
% =========================================================
clc; clear; close all;

%% ===== 1. THÔNG SỐ HÌNH HỌC =====
% [GIU NGUYEN]
H      = 50;                        % cạnh tam giác cơ sở (cm)
l      = 20;                        % chiều dài mỗi khâu (cm)
m_plat = H / 10;                    % cạnh platform (cm)
r      = m_plat / sqrt(3);          % bán kính tâm platform (cm)

Lm = l / 100;                       % chiều dài khâu (m)
rm = r / 100;                       % bán kính platform (m)

% Tọa độ khớp cơ sở (m)
A = [0,    0;
     H,    0;
     H/2,  (sqrt(3)/2)*H] / 100;

% Góc offset platform (rad)
OFFSET_plat = [210; 330; 90] * pi / 180;

%% ===== 2. THÔNG SỐ VẬT LÝ =====
% [DO DAC] - đo bằng cân điện tử
m_link_chu_dong = 0.100;            % kg - khâu chủ động (màu xanh)
m_link_bi_dong  = 0.050;           % kg - khâu bị động  (màu trắng)
m_p             = 0.100;           % kg - platform (cân thực tế)

% Moment quán tính quy về khớp chủ động (kg.m²)
I_chu_dong = (1/3) * m_link_chu_dong * Lm^2;
I_bi_dong  = (1/3) * m_link_bi_dong  * Lm^2;
I_link     = I_chu_dong + I_bi_dong;

% [DO DAC] Ma sát Coulomb (N.m)
% Cách đo: kéo khâu bằng lực kế → lực × cánh tay đòn
tau_friction = 0.005;

% [DO DAC] Ma sát nhớt (N.m.s/rad)
% Cách đo: cho motor quay nhiều tốc độ → đo moment → hệ số góc
B_viscous = 0.001;

% [GIU NGUYEN] Moment tối đa động cơ (N.m)
tau_max = 0.6;

% [GIU NGUYEN] Độ phân giải encoder (xung/vòng) - từ RRResp.ino
CPR = 8192;

% [DO DAC] Nhiễu encoder (rad)
% Cách đo: giữ robot đứng yên, log encoder 10s, tính std
sigma_encoder = 1 * (2*pi / CPR);  % ≈ 0.00077 rad ≈ 0.044°

fprintf('=== THÔNG SỐ ROBOT 3-RRR ===\n');
fprintf('[GIU NGUYEN] L=%.0f cm | H=%.0f cm\n', l, H);
fprintf('[DO DAC] m_link_chu_dong=%.0f g | m_p=%.0f g\n', ...
        m_link_chu_dong*1000, m_p*1000);
fprintf('[TINH] I_link=%.6f kg.m²\n', I_link);
fprintf('[DO DAC] tau_friction=%.4f N.m | B_viscous=%.4f\n', ...
        tau_friction, B_viscous);
fprintf('[DO DAC] sigma_encoder=%.5f rad (%.4f°)\n', ...
        sigma_encoder, rad2deg(sigma_encoder));

%% ===== 3. THÔNG SỐ BỘ ĐIỀU KHIỂN =====
% Quy đổi Kp/Ki/Kd từ ESP32 (đơn vị xung) sang mô phỏng (đơn vị rad)
% Kp_esp32 = 3.5, Ki_esp32 = 0.2, Kd_esp32 = 0.035
scale  = CPR / (2*pi);             % xung/rad ≈ 1303.8
kp_pid = (3.5  / scale) * I_link * 1000;
ki_pid = (0.2  / scale) * I_link * 1000;
kd_pid = (0.035/ scale) * I_link * 1000;

% [TUNING] CTC - thiết kế từ cực vòng kín
% Phương trình đặc trưng: s² + Kd*s + Kp = 0
% Cực: s = -alpha ± j*beta → Kp = alpha²+beta², Kd = 2*alpha
% Muốn ổn định trong Ts giây → alpha ≈ 5/Ts
alpha  = 15;                        % [TUNING] hệ số tắt dần (1/s)
beta   = 10;                        % [TUNING] tần số dao động (rad/s)
Kp_ctc = (alpha^2 + beta^2) * eye(3);   % = 325
Kd_ctc = 2 * alpha            * eye(3); % = 30
Ki_ctc = 5                   * eye(3);  % [TUNING] tăng dần nếu còn sai số tĩnh

fprintf('\n=== THÔNG SỐ BỘ ĐIỀU KHIỂN ===\n');
fprintf('[QUY DOI] PID: Kp=%.5f | Ki=%.6f | Kd=%.7f (N.m/rad)\n', ...
        kp_pid, ki_pid, kd_pid);
fprintf('[TUNING]  CTC: alpha=%d beta=%d → Kp=%.0f Kd=%.0f Ki=%.0f\n', ...
        alpha, beta, Kp_ctc(1,1), Kd_ctc(1,1), Ki_ctc(1,1));

%% ===== 4. CHỌN QUỸ ĐẠO =====
% [GIU NGUYEN] - giống Matlab.txt gốc
fprintf('\n1=Đường thẳng  2=Tam giác  3=Hình vuông  4=Đường tròn\n');
traj_choice = input('Chọn quỹ đạo (1-4): ');
v_des = input('Vận tốc (cm/s): ');

x0 = (H/2) / 100;
y0 = (sqrt(3)/4*H) / 100;
n  = 200;

switch traj_choice
    case 1
        xt = input('X đích (cm): ') / 100;
        yt = input('Y đích (cm): ') / 100;
        dist  = sqrt((xt-x0)^2 + (yt-y0)^2);
        T_sim = dist / (v_des/100);
        x_traj = linspace(x0, xt, n);
        y_traj = linspace(y0, yt, n);

    case 2
        sl = input('Cạnh tam giác (cm): ') / 100;
        ang = [90,210,330]*pi/180;  r_tri = sl/sqrt(3);
        nps = floor(n/3);  x_traj = [];  y_traj = [];
        for i = 1:3
            ni = mod(i,3)+1;
            x_traj = [x_traj, linspace(x0+r_tri*cos(ang(i)), x0+r_tri*cos(ang(ni)), nps)]; %#ok
            y_traj = [y_traj, linspace(y0+r_tri*sin(ang(i)), y0+r_tri*sin(ang(ni)), nps)]; %#ok
        end
        n = length(x_traj);  dist = 3*sl;  T_sim = dist/(v_des/100);

    case 3
        sl = input('Cạnh hình vuông (cm): ') / 100;
        hs = sl/2;
        corn = [x0-hs,y0-hs; x0+hs,y0-hs; x0+hs,y0+hs; x0-hs,y0+hs];
        nps = floor(n/4);  x_traj = [];  y_traj = [];
        for i = 1:4
            ni = mod(i,4)+1;
            x_traj = [x_traj, linspace(corn(i,1), corn(ni,1), nps)]; %#ok
            y_traj = [y_traj, linspace(corn(i,2), corn(ni,2), nps)]; %#ok
        end
        n = length(x_traj);  dist = 4*sl;  T_sim = dist/(v_des/100);

    case 4
        rc = input('Bán kính (cm): ') / 100;
        thc = linspace(0, 2*pi, n);
        x_traj = x0 + rc*cos(thc);
        y_traj = y0 + rc*sin(thc);
        dist = 2*pi*rc;  T_sim = dist/(v_des/100);

    otherwise
        error('Lựa chọn không hợp lệ');
end

T_vec = linspace(0, T_sim, n);
dt    = T_sim / (n-1);
traj_names = {'Đường thẳng','Tam giác','Hình vuông','Đường tròn'};

%% ===== 5. NGHỊCH ĐỘNG HỌC → q_d =====
q_d = zeros(n, 3);
for k = 1:n
    xP = x_traj(k);  yP = y_traj(k);
    for i = 1:3
        Ppx = xP + rm * cos(OFFSET_plat(i));
        Ppy = yP + rm * sin(OFFSET_plat(i));
        dx  = Ppx - A(i,1);
        dy  = Ppy - A(i,2);
        d   = sqrt(dx^2 + dy^2);
        if d <= 2*Lm && d > 1e-4
            q_d(k,i) = atan2(dy,dx) - acos(d/(2*Lm));
        elseif k > 1
            q_d(k,i) = q_d(k-1,i);   % giữ góc trước nếu ngoài workspace
        end
    end
end

% Vận tốc & gia tốc mong muốn (sai phân số)
dq_d  = zeros(n,3);
ddq_d = zeros(n,3);
for i = 1:3
    dq_d(:,i)  = gradient(q_d(:,i),  dt);
    ddq_d(:,i) = gradient(dq_d(:,i), dt);
end

%% ===== 6. MÔ HÌNH ĐỘNG LỰC HỌC M, C, G =====
%
%   M(q)*q̈ + C(q,q̇)*q̇ + G(q) + F(q̇) = τ
%
% M - hằng số (xấp xỉ tuyến tính, đủ dùng khi tốc độ thấp)
M_val  = 2*I_link + (1/3)*m_p*Lm^2;
calc_M = @(q) M_val * eye(3);

% C ≈ 0 khi tốc độ thấp (< 30 RPM)
calc_C = @(q, dq) zeros(3,3);

% G = 0 vì ROBOT NẰM NGANG (mặt phẳng XY, như trong ảnh CAD)
% Trọng lực vuông góc với mặt phẳng chuyển động → không sinh moment
% Nếu đổi robot sang đặt đứng, thay bằng:
%   G_coeff = m_link_chu_dong*9.81*(Lm/2) + m_p*9.81*Lm/3;
%   calc_G  = @(q) G_coeff * cos(q(:));
calc_G = @(q) zeros(3,1);

%% ===== 7. MÔ PHỎNG HAI BỘ ĐIỀU KHIỂN =====
% Trạng thái ban đầu
q_pid  = q_d(1,:)';   dq_pid = zeros(3,1);
q_ctc  = q_d(1,:)';   dq_ctc = zeros(3,1);

% Mảng lưu kết quả
q_pid_log   = zeros(n,3);   q_ctc_log   = zeros(n,3);
tau_pid_log = zeros(n,3);   tau_ctc_log = zeros(n,3);
e_pid_log   = zeros(n,3);   e_ctc_log   = zeros(n,3);

q_pid_log(1,:) = q_pid';
q_ctc_log(1,:) = q_ctc';

% Tích phân và sai số trước
I_err_pid  = zeros(3,1);
I_err_ctc  = zeros(3,1);
e_prev_pid = zeros(3,1);

rng(42);   % seed cố định → kết quả tái hiện được

fprintf('\n=== BẮT ĐẦU MÔ PHỎNG ===\n');

for k = 2:n
    q_ref   = q_d(k,:)';
    dq_ref  = dq_d(k,:)';
    ddq_ref = ddq_d(k,:)';

    % [DO DAC] Nhiễu encoder thực tế
    noise_pid = sigma_encoder * randn(3,1);
    noise_ctc = sigma_encoder * randn(3,1);

    q_meas_pid = q_pid + noise_pid;
    q_meas_ctc = q_ctc + noise_ctc;

    % ---- (A) PID THƯỜNG ----
    e_pid = q_ref - q_meas_pid;

    % Anti-windup: chỉ tích lũy khi chưa bão hòa moment
    tau_test = kp_pid*e_pid + ki_pid*I_err_pid;
    if all(abs(tau_test) < tau_max)
        I_err_pid = I_err_pid + e_pid * dt;
    end

    de_pid    = (e_pid - e_prev_pid) / dt;
    e_prev_pid = e_pid;

    tau_pid = kp_pid*e_pid + ki_pid*I_err_pid + kd_pid*de_pid;
    tau_pid = sign(tau_pid) .* min(abs(tau_pid), tau_max);   % giới hạn moment

    M_k = calc_M(q_pid);
    G_k = calc_G(q_pid);
    friction_pid = tau_friction*sign(dq_pid) + B_viscous*dq_pid;

    ddq_pid = M_k \ (tau_pid - G_k - friction_pid);
    dq_pid  = dq_pid + ddq_pid * dt;
    q_pid   = q_pid  + dq_pid  * dt;

    % ---- (B) COMPUTED TORQUE CONTROL ----
    e_ctc     = q_ref - q_meas_ctc;
    I_err_ctc = I_err_ctc + e_ctc * dt;
    de_ctc    = dq_ref - dq_ctc;

    % Gia tốc mong muốn có bù
    a_des = ddq_ref + Kd_ctc*de_ctc + Kp_ctc*e_ctc + Ki_ctc*I_err_ctc;

    M_c = calc_M(q_ctc);
    C_c = calc_C(q_ctc, dq_ctc);
    G_c = calc_G(q_ctc);

    tau_ctc = M_c*a_des + C_c*dq_ctc + G_c;
    tau_ctc = sign(tau_ctc) .* min(abs(tau_ctc), tau_max);

    friction_ctc = tau_friction*sign(dq_ctc) + B_viscous*dq_ctc;

    ddq_ctc = M_c \ (tau_ctc - C_c*dq_ctc - G_c - friction_ctc);
    dq_ctc  = dq_ctc + ddq_ctc * dt;
    q_ctc   = q_ctc  + dq_ctc  * dt;

    % Lưu kết quả
    q_pid_log(k,:)   = q_pid';     q_ctc_log(k,:)   = q_ctc';
    tau_pid_log(k,:) = tau_pid';   tau_ctc_log(k,:) = tau_ctc';
    e_pid_log(k,:)   = e_pid';     e_ctc_log(k,:)   = e_ctc';
end

fprintf('=== MÔ PHỎNG HOÀN THÀNH ===\n');

%% ===== 8. CHỈ SỐ ĐÁNH GIÁ =====
rms_pid       = sqrt(mean(e_pid_log.^2));
rms_ctc       = sqrt(mean(e_ctc_log.^2));
rms_pid_total = sqrt(mean(e_pid_log(:).^2));
rms_ctc_total = sqrt(mean(e_ctc_log(:).^2));
tau_pid_max   = max(abs(tau_pid_log));
tau_ctc_max   = max(abs(tau_ctc_log));

fprintf('\n=== KẾT QUẢ SO SÁNH ===\n');
fprintf('%-22s %10s %10s %10s %10s\n','','Khớp 1','Khớp 2','Khớp 3','Tổng');
fprintf('%-22s %10.5f %10.5f %10.5f %10.5f\n', ...
        'RMS PID (rad)', rms_pid(1),rms_pid(2),rms_pid(3),rms_pid_total);
fprintf('%-22s %10.5f %10.5f %10.5f %10.5f\n', ...
        'RMS CTC (rad)', rms_ctc(1),rms_ctc(2),rms_ctc(3),rms_ctc_total);
fprintf('%-22s %10.4f %10.4f %10.4f\n', ...
        'Tau max PID (N.m)', tau_pid_max(1),tau_pid_max(2),tau_pid_max(3));
fprintf('%-22s %10.4f %10.4f %10.4f\n', ...
        'Tau max CTC (N.m)', tau_ctc_max(1),tau_ctc_max(2),tau_ctc_max(3));
fprintf('Cải thiện RMS: %.1f%%\n', (1-rms_ctc_total/rms_pid_total)*100);

fprintf('\n--- HƯỚNG DẪN ĐỌC KẾT QUẢ ---\n');
fprintf('RMS PID quá lớn so thực tế → điều chỉnh kp_pid (mục 3)\n');
fprintf('CTC vẫn sai số lớn         → kiểm tra lại calc_G (mục 6)\n');
fprintf('Moment vượt tau_max nhiều  → giảm alpha (mục 3)\n');

%% ===== 9. VẼ ĐỒ THỊ SO SÁNH =====
col_pid  = [0.85, 0.15, 0.15];
col_ctc  = [0.10, 0.45, 0.85];
col_ref  = [0.10, 0.10, 0.10];
khop_name = {'Khớp 1','Khớp 2','Khớp 3'};

fig1 = figure('Name','So Sánh PID vs CTC','NumberTitle','off', ...
              'Color','white','Units','normalized','OuterPosition',[0 0 1 1]);

% Hàng 1: Góc khớp
for i = 1:3
    ax = subplot(3,3,i); hold on; grid on; box on;
    plot(T_vec, rad2deg(q_d(:,i)),       '--','Color',col_ref,'LineWidth',1.5,'DisplayName','Tham chiếu');
    plot(T_vec, rad2deg(q_pid_log(:,i)), '-', 'Color',col_pid,'LineWidth',1.8,'DisplayName','PID thường');
    plot(T_vec, rad2deg(q_ctc_log(:,i)), '-', 'Color',col_ctc,'LineWidth',1.8,'DisplayName','CTC (bậc 2)');
    title(sprintf('%s - Góc khớp', khop_name{i}),'FontWeight','bold');
    xlabel('Thời gian (s)'); ylabel('Góc (độ)');
    if i == 1, legend('Location','best','FontSize',8); end
    set(ax,'FontSize',9);
end

% Hàng 2: Sai số góc
for i = 1:3
    ax = subplot(3,3,3+i); hold on; grid on; box on;
    plot(T_vec, rad2deg(e_pid_log(:,i)), '-','Color',col_pid,'LineWidth',1.8,'DisplayName','PID thường');
    plot(T_vec, rad2deg(e_ctc_log(:,i)), '-','Color',col_ctc,'LineWidth',1.8,'DisplayName','CTC (bậc 2)');
    yline(0,'k--','LineWidth',0.8,'HandleVisibility','off');
    title(sprintf('%s - Sai số', khop_name{i}),'FontWeight','bold');
    xlabel('Thời gian (s)'); ylabel('Sai số (độ)');
    text(0.02,0.92, sprintf('RMS: PID=%.4f° CTC=%.4f°', ...
         rad2deg(rms_pid(i)), rad2deg(rms_ctc(i))), ...
         'Units','normalized','FontSize',8,'Color',[0.3 0.3 0.3]);
    if i == 1, legend('Location','best','FontSize',8); end
    set(ax,'FontSize',9);
end

% Hàng 3: Moment
for i = 1:3
    ax = subplot(3,3,6+i); hold on; grid on; box on;
    plot(T_vec, tau_pid_log(:,i)*1000, '-','Color',col_pid,'LineWidth',1.8,'DisplayName','PID thường');
    plot(T_vec, tau_ctc_log(:,i)*1000, '-','Color',col_ctc,'LineWidth',1.8,'DisplayName','CTC (bậc 2)');
    yline( tau_max*1000,'r--','LineWidth',1,'HandleVisibility','off');
    yline(-tau_max*1000,'r--','LineWidth',1,'HandleVisibility','off');
    title(sprintf('%s - Moment điều khiển', khop_name{i}),'FontWeight','bold');
    xlabel('Thời gian (s)'); ylabel('Moment (mN.m)');
    text(0.02,0.92, sprintf('Max: PID=%.1f CTC=%.1f mN.m', ...
         tau_pid_max(i)*1000, tau_ctc_max(i)*1000), ...
         'Units','normalized','FontSize',8,'Color',[0.3 0.3 0.3]);
    if i == 1, legend('Location','best','FontSize',8); end
    set(ax,'FontSize',9);
end

sgtitle(fig1, sprintf('So Sánh PID Thường vs CTC | %s | v=%.1f cm/s | T=%.2f s | Cải thiện: %.1f%%', ...
        traj_names{traj_choice}, v_des, T_sim, ...
        (1-rms_ctc_total/rms_pid_total)*100), ...
        'FontSize',12,'FontWeight','bold');

%% ===== 10. HÌNH PHÂN TÍCH TỔNG HỢP =====
fig2 = figure('Name','Phân Tích Tổng Hợp','NumberTitle','off', ...
              'Color','white','Units','normalized','OuterPosition',[0.05 0.05 0.90 0.90]);

% Sai số tổng norm
ax_e = subplot(2,2,[1,2]);
hold on; grid on; box on;
norm_e_pid = sqrt(sum(e_pid_log.^2, 2));
norm_e_ctc = sqrt(sum(e_ctc_log.^2, 2));
plot(T_vec, rad2deg(norm_e_pid), '-','Color',col_pid,'LineWidth',2,'DisplayName','PID thường');
plot(T_vec, rad2deg(norm_e_ctc), '-','Color',col_ctc,'LineWidth',2,'DisplayName','CTC (bậc 2)');
title('Sai số tổng 3 khớp ||e(t)||','FontSize',12,'FontWeight','bold');
xlabel('Thời gian (s)'); ylabel('||e|| (độ)');
legend('Location','best','FontSize',10);

% Bar chart RMS
ax_b = subplot(2,2,3);
bar_data = [rad2deg(rms_pid); rad2deg(rms_ctc)]';
b = bar(ax_b, bar_data, 'grouped');
b(1).FaceColor = col_pid;
b(2).FaceColor = col_ctc;
set(ax_b,'XTickLabel',khop_name,'FontSize',10);
title('RMS Sai Số Từng Khớp','FontSize',11,'FontWeight','bold');
ylabel('RMS (độ)');
legend(ax_b, {'PID thường','CTC (bậc 2)'},'Location','best','FontSize',9);
grid(ax_b,'on');

% Bảng tổng kết
ax_t = subplot(2,2,4);
axis off; xlim([0 1]); ylim([0 1]);

improve_rms = (1 - rms_ctc_total/rms_pid_total)*100;
headers = {'Chỉ số','PID Thường','CTC Bậc 2','Cải thiện'};
rows = {
    'RMS tổng (độ)',    sprintf('%.4f',rad2deg(rms_pid_total)), ...
                        sprintf('%.4f',rad2deg(rms_ctc_total)), ...
                        sprintf('%.1f%%',improve_rms);
    'τ max K1 (mN.m)',  sprintf('%.1f',tau_pid_max(1)*1000), ...
                        sprintf('%.1f',tau_ctc_max(1)*1000), '-';
    'τ max K2 (mN.m)',  sprintf('%.1f',tau_pid_max(2)*1000), ...
                        sprintf('%.1f',tau_ctc_max(2)*1000), '-';
    'τ max K3 (mN.m)',  sprintf('%.1f',tau_pid_max(3)*1000), ...
                        sprintf('%.1f',tau_ctc_max(3)*1000), '-';
    'Kp',               sprintf('%.5f',kp_pid),  sprintf('%.0f',Kp_ctc(1,1)), '-';
    'Ki',               sprintf('%.6f',ki_pid),  sprintf('%.0f',Ki_ctc(1,1)), '-';
    'Kd',               sprintf('%.7f',kd_pid),  sprintf('%.0f',Kd_ctc(1,1)), '-';
    'tau_friction(N.m)',sprintf('%.4f',tau_friction), '[DO DAC - dùng lực kế]', '-';
    'sigma encoder(°)', sprintf('%.4f',rad2deg(sigma_encoder)), '[DO DAC - log 10s]', '-';
};

col_x = [0.01, 0.30, 0.55, 0.78];
col_w = 0.26;
row_tops = linspace(0.92, 0.02, size(rows,1)+1);
row_h    = 0.25;
hdr_bg   = [0.15 0.35 0.65];

for c = 1:4
    rectangle(ax_t,'Position',[col_x(c), row_tops(1)-row_h, col_w-0.004, row_h-0.02], ...
