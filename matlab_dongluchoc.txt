function PID_DongLucHoc
% =========================================================
%  BỘ ĐIỀU KHIỂN PID ĐỘNG LỰC HỌC BẬC 2 - ROBOT 3-RRR
%  Computed Torque Control (CTC) vs PID thường
%
%  Các chỗ cần đo đạc thực tế được đánh dấu:
%  [DO DAC] = bạn cần đo/tính từ phần cứng thực tế
%  [TUNING] = bạn cần chỉnh sau khi có kết quả thực tế
%  [GIU NGUYEN] = lấy từ file Matlab.txt gốc, không cần đổi
% =========================================================
clc; clear; close all;

%% ========== 1. THÔNG SỐ HÌNH HỌC ROBOT ==========
% [GIU NGUYEN] - lấy từ file Matlab.txt gốc

H      = 50;             % cạnh tam giác cơ sở (cm)
l      = 20;             % chiều dài mỗi khâu (cm)
m_plat = H / 10;         % cạnh platform (cm)
r      = m_plat / sqrt(3); % bán kính tâm platform (cm)

% Chuyển sang đơn vị SI (m)
Lm = l / 100;
rm = r / 100;

% Tọa độ khớp cơ sở (m)
A = [0,       0;
     H,       0;
     H/2,     (sqrt(3)/2)*H] / 100;

% Góc offset platform
OFFSET_plat = [210; 330; 90] * pi / 180;

% Offset góc khớp (từ RRResp.ino)
% [GIU NGUYEN]
offset_deg = [-19.49; 100.51; -139.49];

%% ========== 2. THÔNG SỐ VẬT LÝ - CẦN ĐO ĐẠC ==========

% ----------------------------------------------------------
% [DO DAC] KHỐI LƯỢNG TỪNG KHÂU (kg)
% Cách đo: dùng cân điện tử cân từng thanh khâu riêng lẻ
% Hiện tại đang dùng: 100g = 0.1 kg (lấy từ Matlab.txt)
% ----------------------------------------------------------
m_link = 0.100;          % kg - THAY bằng giá trị cân được

% ----------------------------------------------------------
% [DO DAC] KHỐI LƯỢNG PLATFORM (kg)
% Cách đo: cân toàn bộ tam giác platform + các khớp bị động
% Hiện tại đang dùng: 100g = 0.1 kg
% ----------------------------------------------------------
m_p = 0.100;             % kg - THAY bằng giá trị cân được

% ----------------------------------------------------------
% [DO DAC] MOMENT QUÁN TÍNH KHÂU I_link (kg.m²)
% Công thức lý thuyết (thanh đồng nhất quay quanh đầu):
%   I = (1/3)*m*L²
% Cách đo thực tế (phương pháp dao động):
%   1. Treo khâu bằng dây ở một đầu
%   2. Cho dao động như con lắc vật lý
%   3. Đo chu kỳ T (s)
%   4. I = m*g*(L/2)*T²/(4*pi²)
% Hiện tại dùng công thức lý thuyết:
% ----------------------------------------------------------
I_link = (1/3) * m_link * Lm^2;
% I_link = ???;          % THAY nếu đã đo được

% ----------------------------------------------------------
% [DO DAC] MA SÁT COULOMB (N.m)
% Cách đo:
%   1. Cho motor quay chậm không tải (PWM thấp nhất còn quay)
%   2. Đọc PWM đó → quy ra voltage → quy ra moment qua Kt
%   3. Hoặc: kéo tay quay khâu bằng lực kế, đọc lực × cánh tay đòn
% Hiện tại ước lượng: 0.005 N.m (nhỏ)
% ----------------------------------------------------------
tau_friction = 0.005;    % N.m - THAY bằng giá trị đo được

% ----------------------------------------------------------
% [DO DAC] MA SÁT NHỚT (N.m.s/rad)
% Cách đo:
%   1. Cho motor quay ở nhiều tốc độ khác nhau (ổn định)
%   2. Đo moment giữ tốc độ tại mỗi tốc độ → vẽ đồ thị tau vs omega
%   3. Hệ số góc đường thẳng = B (hệ số ma sát nhớt)
% Hiện tại ước lượng: 0.001
% ----------------------------------------------------------
B_viscous = 0.001;       % N.m.s/rad - THAY bằng giá trị đo được

% ----------------------------------------------------------
% [DO DAC] MOMENT TỐI ĐA ĐỘNG CƠ (N.m)
% Cách đo:
%   1. Xem datasheet động cơ → lấy stall torque
%   2. Hoặc: giữ chặt trục, đo dòng stall → tau = Kt * I_stall
%      (Kt thường ghi trong datasheet, đơn vị N.m/A)
% [GIU NGUYEN] - lấy từ Matlab.txt gốc
% ----------------------------------------------------------
tau_max = 0.6;           % N.m

% ----------------------------------------------------------
% [DO DAC] ĐỘ PHÂN GIẢI ENCODER (xung/vòng)
% Cách xác định:
%   1. Xem datasheet encoder → CPR (counts per revolution)
%   2. Hoặc: quay đúng 1 vòng tay, đếm xung → đó là CPR
%   3. Với quadrature encoder: CPR = PPR * 4
% [GIU NGUYEN] - lấy từ RRResp.ino
% ----------------------------------------------------------
CPR = 8192;              % xung/vòng

% ----------------------------------------------------------
% [DO DAC] NHIỄU ENCODER (rad)
% Cách đo:
%   1. Giữ robot đứng yên hoàn toàn
%   2. Log dữ liệu encoder 10 giây
%   3. Tính độ lệch chuẩn → đó là sigma_encoder
%   4. Quy ra rad: sigma_rad = sigma_xung * 2*pi / CPR
% Hiện tại ước lượng 1 xung encoder:
% ----------------------------------------------------------
sigma_encoder = 1 * (2*pi / CPR);  % rad (~0.00077 rad ≈ 0.044°)
% sigma_encoder = ???;  % THAY bằng giá trị đo được

fprintf('=== THÔNG SỐ ROBOT 3-RRR ===\n');
fprintf('[GIU NGUYEN] L=%.0f cm | H=%.0f cm\n', l, H);
fprintf('[DO DAC] m_link=%.0f g | m_plat=%.0f g\n', m_link*1000, m_p*1000);
fprintf('[TINH]   I_link=%.6f kg.m²\n', I_link);
fprintf('[DO DAC] tau_friction=%.4f N.m | B_viscous=%.4f\n', tau_friction, B_viscous);
fprintf('[DO DAC] sigma_encoder=%.5f rad (%.4f°)\n', sigma_encoder, rad2deg(sigma_encoder));

%% ========== 3. THÔNG SỐ BỘ ĐIỀU KHIỂN ==========

% ----------------------------------------------------------
% [TUNING] HỆ SỐ PID THƯỜNG - QUY ĐỔI TỪ ESP32
%
% Trên ESP32, PID hoạt động trên đơn vị XUNG encoder:
%   output = Kp*e_xung + Ki*∫e_xung + Kd*de_xung
%
% Trong mô phỏng, hoạt động trên đơn vị RAD:
%   Cần quy đổi: Kp_sim = Kp_esp32 / scale
%   với scale = CPR / (2*pi)  [xung/rad]
%
% Cách chỉnh Kp/Ki/Kd thực tế (phương pháp Ziegler-Nichols):
%   1. Đặt Ki=0, Kd=0, tăng Kp dần
%   2. Tìm Kp_ultimate (Ku) = Kp khi robot bắt đầu dao động liên tục
%   3. Đo chu kỳ dao động Tu (giây)
%   4. Kp = 0.6*Ku | Ki = 1.2*Ku/Tu | Kd = 0.075*Ku*Tu
%
% Giá trị từ RRResp.ino: Kp=3.5 | Ki=0.2 | Kd=0.035 (đơn vị xung)
% ----------------------------------------------------------
scale  = CPR / (2*pi);   % xung/rad ≈ 1303.8

% Quy đổi về N.m/rad để dùng trong mô phỏng động lực học
% Nhân thêm I_link để đổi từ "tỉ lệ góc" sang "moment lực"
kp_pid = (3.5  / scale) * I_link * 1000;
ki_pid = (0.2  / scale) * I_link * 1000;
kd_pid = (0.035/ scale) * I_link * 1000;
% [TUNING] Nếu kết quả mô phỏng quá khác thực tế:
%   - PID hội tụ quá chậm → tăng kp_pid
%   - PID dao động nhiều  → giảm kp_pid hoặc tăng kd_pid
%   - PID còn sai số tĩnh → tăng ki_pid

% ----------------------------------------------------------
% [TUNING] HỆ SỐ CTC - THIẾT KẾ TỪ CỰC VÒNG KÍN
%
% Phương trình đặc trưng vòng kín CTC:
%   s² + Kd*s + Kp = 0
%   Cực: s = -alpha ± j*beta
%   → Kp = alpha² + beta²
%   → Kd = 2*alpha
%
% Cách chọn alpha, beta:
%   - alpha = hệ số tắt dần → càng lớn hội tụ càng nhanh
%     Gợi ý: alpha = 3~5 / T_settle (T_settle là thời gian ổn định mong muốn)
%     Ví dụ muốn ổn định trong 0.3s: alpha = 5/0.3 ≈ 17
%   - beta = tần số dao động tắt dần
%     Gợi ý: beta = 0.5~1.0 * alpha (dao động tắt nhanh)
%   - Kiểm tra: omega_n = sqrt(alpha²+beta²) < omega_max_motor
%     omega_max_motor = RPM_max * 2*pi/60
%
% Hiện tại: alpha=15, beta=10 → Kp=325, Kd=30
% [TUNING] Nếu robot thực tế bị rung → giảm alpha
%          Nếu robot phản ứng quá chậm → tăng alpha
% ----------------------------------------------------------
alpha = 15;   % [TUNING] hệ số tắt dần - đơn vị: 1/s
beta  = 10;   % [TUNING] tần số dao động - đơn vị: rad/s

Kp_ctc = (alpha^2 + beta^2) * eye(3);   % = 325
Kd_ctc = 2 * alpha * eye(3);            % = 30
Ki_ctc = 5 * eye(3);
% [TUNING] Ki_ctc: bắt đầu từ 0, tăng dần nếu còn sai số tĩnh

fprintf('\n=== THÔNG SỐ BỘ ĐIỀU KHIỂN ===\n');
fprintf('[QUY DOI] PID: Kp=%.5f | Ki=%.6f | Kd=%.7f (N.m/rad)\n', kp_pid, ki_pid, kd_pid);
fprintf('[TUNING]  CTC: alpha=%d beta=%d → Kp=%.0f Kd=%.0f Ki=%.0f\n', ...
    alpha, beta, Kp_ctc(1,1), Kd_ctc(1,1), Ki_ctc(1,1));

%% ========== 4. CHỌN QUỸ ĐẠO ==========
% [GIU NGUYEN] - giống Matlab.txt gốc

fprintf('\n1=Đường thẳng  2=Tam giác  3=Hình vuông  4=Đường tròn\n');
traj_choice = input('Chọn quỹ đạo (1-4): ');
v_des = input('Vận tốc (cm/s): ');

x0 = (H/2) / 100;
y0 = (sqrt(3)/4*H) / 100;
n  = 200;

switch traj_choice
    case 1
        xt = input('X đích (cm): ') / 100;
        yt = input('Y đích (cm): ') / 100;
        dist = sqrt((xt-x0)^2 + (yt-y0)^2);
        T_sim = dist / (v_des/100);
        x_traj = linspace(x0, xt, n);
        y_traj = linspace(y0, yt, n);
    case 2
        sl = input('Cạnh tam giác (cm): ') / 100;
        ang = [90,210,330]*pi/180;  r_tri = sl/sqrt(3);
        nps = floor(n/3);  x_traj = [];  y_traj = [];
        for i=1:3
            ni=mod(i,3)+1;
            x_traj=[x_traj, linspace(x0+r_tri*cos(ang(i)), x0+r_tri*cos(ang(ni)), nps)];
            y_traj=[y_traj, linspace(y0+r_tri*sin(ang(i)), y0+r_tri*sin(ang(ni)), nps)];
        end
        n=length(x_traj);  dist=3*sl;  T_sim=dist/(v_des/100);
    case 3
        sl = input('Cạnh hình vuông (cm): ') / 100;
        hs=sl/2;
        corn=[x0-hs,y0-hs; x0+hs,y0-hs; x0+hs,y0+hs; x0-hs,y0+hs];
        nps=floor(n/4);  x_traj=[];  y_traj=[];
        for i=1:4
            ni=mod(i,4)+1;
            x_traj=[x_traj, linspace(corn(i,1),corn(ni,1),nps)];
            y_traj=[y_traj, linspace(corn(i,2),corn(ni,2),nps)];
        end
        n=length(x_traj);  dist=4*sl;  T_sim=dist/(v_des/100);
    case 4
        rc = input('Bán kính (cm): ') / 100;
        thc=linspace(0,2*pi,n);
        x_traj=x0+rc*cos(thc);  y_traj=y0+rc*sin(thc);
        dist=2*pi*rc;  T_sim=dist/(v_des/100);
    otherwise
        error('Lựa chọn không hợp lệ');
end

T_vec = linspace(0, T_sim, n);
dt    = T_sim / (n-1);
traj_names = {'Đường thẳng','Tam giác','Hình vuông','Đường tròn'};

%% ========== 5. TÍNH NGHỊCH ĐỘNG HỌC → q_d ==========

q_d = zeros(n, 3);
for k = 1:n
    xP = x_traj(k);  yP = y_traj(k);
    for i = 1:3
        Ppx = xP + rm * cos(OFFSET_plat(i));
        Ppy = yP + rm * sin(OFFSET_plat(i));
        dx = Ppx - A(i,1);  dy = Ppy - A(i,2);
        d  = sqrt(dx^2 + dy^2);
        if d <= 2*Lm && d > 1e-4
            q_d(k,i) = atan2(dy,dx) - acos(d/(2*Lm));
        elseif k > 1
            q_d(k,i) = q_d(k-1,i);
        end
    end
end

% Vận tốc và gia tốc mong muốn (sai phân số)
dq_d  = zeros(n,3);
ddq_d = zeros(n,3);
for i = 1:3
    dq_d(:,i)  = gradient(q_d(:,i),  dt);
    ddq_d(:,i) = gradient(dq_d(:,i), dt);
end

%% ========== 6. MA TRẬN ĐỘNG LỰC HỌC M, C, G ==========
%
% Mô hình đơn giản hóa cho mỗi chuỗi động học độc lập:
%
%   M(q)*q̈ + C(q,q̇)*q̇ + G(q) + F(q̇) = τ
%
% ----------------------------------------------------------
% [DO DAC] MA TRẬN QUÁN TÍNH M (kg.m²)
%   Hiện tại: M = hằng số (xấp xỉ tuyến tính)
%   Thực tế M phụ thuộc vào cấu hình q → thay đổi theo thời gian
%   Để có M chính xác hơn: dùng phần mềm CAD (SolidWorks/Fusion360)
%   → Export moment of inertia tại nhiều cấu hình → fit hàm M(q)
% ----------------------------------------------------------
M_val = 2*I_link + (1/3)*m_p*Lm^2;
calc_M = @(q) M_val * eye(3);

% ----------------------------------------------------------
% [DO DAC] MA TRẬN CORIOLIS C (N.m.s/rad)
%   Hiện tại: C ≈ 0 (bỏ qua - hợp lý khi tốc độ thấp < 30 RPM)
%   Nếu robot chạy nhanh hơn → cần tính C đầy đủ từ Lagrangian
%   Điều kiện bỏ qua C: omega_max < sqrt(G_max / (2*M_val*L))
%   omega_max (rad/s) của bạn = RPM_max * 2*pi/60
% ----------------------------------------------------------
calc_C = @(q, dq) zeros(3,3);

% ----------------------------------------------------------
% [DO DAC / TINH] MA TRẬN TRỌNG LỰC G (N.m)
%   Công thức: G_i = (m_link*g*L/2 + m_p*g*L/3) * cos(q_i)
%   Chú ý: robot 3-RRR nằm ngang (mặt phẳng XY) hay đứng (XZ)?
%   - Nằm ngang (XY): G ≈ 0 vì trọng lực vuông góc mặt chuyển động
%   - Đứng (XZ hoặc YZ): G có ảnh hưởng lớn như công thức dưới
%
%   [QUAN TRONG] Xác định hướng đặt robot của bạn:
%   Nếu robot nằm ngang → đổi calc_G = @(q) zeros(3,1)
%   Nếu robot đứng (như hiện tại) → giữ công thức dưới
% ----------------------------------------------------------
g_acc = 9.81;
G_coeff = m_link*g_acc*(Lm/2) + m_p*g_acc*Lm/3;
calc_G = @(q) G_coeff * cos(q(:));
% Nếu robot nằm ngang, THAY bằng:
% calc_G = @(q) zeros(3,1);

%% ========== 7. MÔ PHỎNG HAI BỘ ĐIỀU KHIỂN ==========

% Khởi tạo trạng thái ban đầu
q_pid  = q_d(1,:)';   dq_pid  = zeros(3,1);
q_ctc  = q_d(1,:)';   dq_ctc  = zeros(3,1);

% Mảng lưu kết quả
q_pid_log   = zeros(n,3);  q_ctc_log   = zeros(n,3);
tau_pid_log = zeros(n,3);  tau_ctc_log = zeros(n,3);
e_pid_log   = zeros(n,3);  e_ctc_log   = zeros(n,3);

q_pid_log(1,:) = q_pid';
q_ctc_log(1,:) = q_ctc';

% Tích phân và sai số trước
I_err_pid  = zeros(3,1);
I_err_ctc  = zeros(3,1);
e_prev_pid = zeros(3,1);

% Seed nhiễu cố định để kết quả tái hiện được
rng(42);

fprintf('\n=== BẮT ĐẦU MÔ PHỎNG ===\n');

for k = 2:n
    q_ref   = q_d(k,:)';
    dq_ref  = dq_d(k,:)';
    ddq_ref = ddq_d(k,:)';

    % ----------------------------------------------------------
    % [DO DAC] NHIỄU ENCODER
    % Đây là nhiễu đo lường từ encoder thực tế
    % Cách đo: giữ robot đứng yên, log encoder 10s, tính std
    % Hiện tại dùng: sigma = 1 xung encoder
    % ----------------------------------------------------------
    noise_pid = sigma_encoder * randn(3,1);
    noise_ctc = sigma_encoder * randn(3,1);

    % Góc đo được (có nhiễu encoder)
    q_meas_pid = q_pid + noise_pid;
    q_meas_ctc = q_ctc + noise_ctc;

    % ---- (A) PID THƯỜNG ----
    e_pid      = q_ref - q_meas_pid;
    I_err_pid  = I_err_pid + e_pid * dt;
    de_pid     = (e_pid - e_prev_pid) / dt;
    e_prev_pid = e_pid;

    % Moment PID: τ = Kp*e + Ki*∫e + Kd*ė
    tau_pid = kp_pid*e_pid + ki_pid*I_err_pid + kd_pid*de_pid;

    % Giới hạn moment theo motor thực tế
    tau_pid = sign(tau_pid) .* min(abs(tau_pid), tau_max);

    % Ma sát thực tế
    % [DO DAC] tau_friction và B_viscous - xem phần 2
    friction_pid = tau_friction*sign(dq_pid) + B_viscous*dq_pid;

    % Động lực học: M*q̈ = τ - G - ma sát
    M_k  = calc_M(q_pid);
    G_k  = calc_G(q_pid);
    ddq_pid = M_k \ (tau_pid - G_k - friction_pid);

    % Tích phân Euler
    dq_pid = dq_pid + ddq_pid * dt;
    q_pid  = q_pid  + dq_pid  * dt;

    % ---- (B) COMPUTED TORQUE CONTROL (PID bậc 2) ----
    e_ctc     = q_ref - q_meas_ctc;
    I_err_ctc = I_err_ctc + e_ctc * dt;
    de_ctc    = dq_ref - dq_ctc;

    % Gia tốc mong muốn có bù:
    % a_d = q̈_ref + Kd*(q̇_ref-q̇) + Kp*(q_ref-q) + Ki*∫e
    a_des = ddq_ref + Kd_ctc*de_ctc + Kp_ctc*e_ctc + Ki_ctc*I_err_ctc;

    % Moment CTC: τ = M*a_des + C*q̇ + G  (bù hoàn toàn động lực học)
    M_c   = calc_M(q_ctc);
    C_c   = calc_C(q_ctc, dq_ctc);
    G_c   = calc_G(q_ctc);
    tau_ctc = M_c*a_des + C_c*dq_ctc + G_c;

    % Giới hạn moment
    tau_ctc = sign(tau_ctc) .* min(abs(tau_ctc), tau_max);

    % Ma sát
    friction_ctc = tau_friction*sign(dq_ctc) + B_viscous*dq_ctc;

    % Động lực học
    ddq_ctc = M_c \ (tau_ctc - C_c*dq_ctc - G_c - friction_ctc);
    dq_ctc  = dq_ctc + ddq_ctc * dt;
    q_ctc   = q_ctc  + dq_ctc  * dt;

    % Lưu kết quả
    q_pid_log(k,:)   = q_pid';
    q_ctc_log(k,:)   = q_ctc';
    tau_pid_log(k,:) = tau_pid';
    tau_ctc_log(k,:) = tau_ctc';
    e_pid_log(k,:)   = e_pid';
    e_ctc_log(k,:)   = e_ctc';
end

fprintf('=== MÔ PHỎNG HOÀN THÀNH ===\n');

%% ========== 8. TÍNH CHỈ SỐ ĐÁNH GIÁ ==========

rms_pid       = sqrt(mean(e_pid_log.^2));
rms_ctc       = sqrt(mean(e_ctc_log.^2));
rms_pid_total = sqrt(mean(e_pid_log(:).^2));
rms_ctc_total = sqrt(mean(e_ctc_log(:).^2));
tau_pid_max   = max(abs(tau_pid_log));
tau_ctc_max   = max(abs(tau_ctc_log));

fprintf('\n=== KẾT QUẢ SO SÁNH ===\n');
fprintf('%-22s %10s %10s %10s %10s\n','','Khớp 1','Khớp 2','Khớp 3','Tổng');
fprintf('%-22s %10.5f %10.5f %10.5f %10.5f\n','RMS PID (rad)', rms_pid(1),rms_pid(2),rms_pid(3),rms_pid_total);
